cmake_minimum_required(VERSION 3.20)
project(porter_rl_dump LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
get_filename_component(ROOT "../.." ABSOLUTE)
file(GLOB PKG "${ROOT}/Unreal/UnrealPackage/UnPackage*.cpp")
file(GLOB FS "${ROOT}/Unreal/FileSystem/*.cpp")
file(GLOB ZLIB "${ROOT}/libs/zlib/*.c")
file(GLOB AES "${ROOT}/libs/rijndael/*.c")
add_executable(porter-rl-dump Main.cpp ${PKG} ${FS} ${ZLIB} ${AES}
${ROOT}/Unreal/UnCore.cpp ${ROOT}/Unreal/UnCoreCompression.cpp ${ROOT}/Unreal/UnCoreDecrypt.cpp ${ROOT}/Unreal/UnCoreSerialize.cpp ${ROOT}/Unreal/UnObject.cpp ${ROOT}/Unreal/UnObject4.cpp ${ROOT}/Unreal/GameDatabase.cpp ${ROOT}/Unreal/TypeInfo.cpp ${ROOT}/Core/Core.cpp ${ROOT}/Core/CoreWin32.cpp ${ROOT}/Core/Memory.cpp ${ROOT}/libs/lzo/lzo_init.c ${ROOT}/libs/lzo/lzo1x_d2.c ${ROOT}/libs/mspack/lzxd.c ${ROOT}/libs/lz4/lz4.c)
target_include_directories(porter-rl-dump PRIVATE . ${ROOT} ${ROOT}/Core ${ROOT}/Unreal ${ROOT}/Unreal/UnrealPackage ${ROOT}/libs ${ROOT}/libs/include ${ROOT}/libs/zlib ${ROOT}/libs/rijndael)
target_compile_options(porter-rl-dump PRIVATE -w -fpermissive -fno-strict-aliasing)
target_compile_definitions(porter-rl-dump PRIVATE UNREAL4=0 USE_OODLE=0)
target_link_options(porter-rl-dump PRIVATE -static -static-libgcc -static-libstdc++)
target_link_libraries(porter-rl-dump PRIVATE dbghelp user32)
